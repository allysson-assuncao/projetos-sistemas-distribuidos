 136 |     # Consenso dinâmico: 3 nós (maioria 2/3 mascara 1 bizantino), 2 nós (2/2), 1 nó (1/1 degradado)
 137 |     @classmethod
 138 |     def calcular_consenso(
 139 |         cls,
 140 |         respostas: list[dict],
 141 |         total_ativos: int | None = None,
 142 |     ) -> dict:
 143 |         """
 144 |         Aplica quórum dinâmico sobre as respostas recebidas dos nós ativos:
 145 |         - 3 ativos: maioria (>= 2 votos) mascara 1 nó bizantino.
 146 |         - 2 ativos: exige concordância (2/2); divergência levanta ErroQuorum.
 147 |         - 1 ativo: modo degradado (1/1) preserva disponibilidade do sistema.
 148 |         - 0 ativos: levanta ErroQuorum.
 149 |         """
 150 |         if not respostas:
 151 |             raise ErroQuorum("ERRO_QUORUM: Nenhum controlador ativo respondeu.")
 152 | 
 153 |         contagem = collections.Counter(cls._normalizar_estado(r) for r in respostas)
 154 |         chave_vencedora, votos = contagem.most_common(1)[0]
 155 |         n_respostas = len(respostas) if total_ativos is None else max(len(respostas), total_ativos)
 156 | 
 157 |         if n_respostas >= 3 and votos < 2:
 158 |             raise ErroQuorum(
 159 |                 f"ERRO_QUORUM: Divergência total entre os {len(respostas)} controladores ativos."
 160 |             )
 161 |         if n_respostas == 2 and votos < 2:
 162 |             raise ErroQuorum(
 163 |                 "ERRO_QUORUM: Conflito entre os 2 controladores ativos (sem desempate)."
 164 |             )

 166 |         for resposta in respostas:
 167 |             if cls._normalizar_estado(resposta) == chave_vencedora:
 168 |                 return dict(resposta)
